        function playBeep(type) {
            if (!soundEnabled) return;
            if (!audioCtx) audioCtx = new AudioContext();
            const osc = audioCtx.createOscillator();
            const gainNode = audioCtx.createGain();
            osc.connect(gainNode); gainNode.connect(audioCtx.destination);
            
            if (type === 'correct') {
                osc.type = 'sine'; osc.frequency.setValueAtTime(880, audioCtx.currentTime); 
                osc.frequency.exponentialRampToValueAtTime(1760, audioCtx.currentTime + 0.1);
                gainNode.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
            } else {
                osc.type = 'sawtooth'; osc.frequency.setValueAtTime(300, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(100, audioCtx.currentTime + 0.3);
                gainNode.gain.setValueAtTime(0.3, audioCtx.currentTime);
                gainNode.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.3);
            }
            osc.start(); osc.stop(audioCtx.currentTime + 0.3);
        }

        function generateValidAShape() {
            let valid = false;
            let resultBase = [], resultTargets = [], sideM = 10, cellAreaM2 = 100;
            let sideOptions = [10, 20, 50, 100, 200, 500, 1000]; 
            
            const shapeMaps = [
                ["111111", "111111", "111111", "111111"], 
                ["11111", "11111", "11111", "11111", "11111"], 
                ["111000", "111000", "111111", "111111"], 
                ["110011", "110011", "111111", "111111"], 
                ["001100", "111111", "111111", "001100"], 
                ["00100", "01110", "11111", "11111"], 
                ["011110", "111111", "111111"], 
                ["01100", "11110", "11011", "01111", "00110"], 
                ["11100", "11111", "01111", "00110"]
            ];

            while(!valid) {
                sideM = sideOptions[Math.floor(Math.random() * sideOptions.length)];
                cellAreaM2 = sideM * sideM;
                
                let baseCells = [];
                let mapStr = shapeMaps[Math.floor(Math.random() * shapeMaps.length)];
                
                for(let y=0; y<mapStr.length; y++) {
                    for(let x=0; x<mapStr[y].length; x++) {
                        if(mapStr[y][x] === '1') {
                            baseCells.push({x, y, region: -1});
                        }
                    }
                }

                let simCells = JSON.parse(JSON.stringify(baseCells));
                simCells.forEach(c => c.region = 0);
                
                let numCuts = Math.floor(Math.random() * 2) + 1;
                let maxReg = 0;

                for(let c=0; c<numCuts; c++) {
                    let targetReg = Math.floor(Math.random() * (maxReg + 1));
                    let pool = simCells.filter(cell => cell.region === targetReg);
                    if(pool.length < 4) continue;
                    
                    let minX = 99, maxX = -1, minY = 99, maxY = -1;
                    pool.forEach(cell => { if(cell.x<minX) minX=cell.x; if(cell.x>maxX) maxX=cell.x; if(cell.y<minY) minY=cell.y; if(cell.y>maxY) maxY=cell.y; });
                    
                    let cutDir = Math.random() < 0.5 ? 'H' : 'V';
                    if(cutDir === 'H' && maxY > minY) {
                        let cutPos = minY + Math.floor(Math.random() * (maxY - minY)) + 1;
                        maxReg++; pool.forEach(cell => { if(cell.y >= cutPos) cell.region = maxReg; });
                    } else if (cutDir === 'V' && maxX > minX) {
                        let cutPos = minX + Math.floor(Math.random() * (maxX - minX)) + 1;
                        maxReg++; pool.forEach(cell => { if(cell.x >= cutPos) cell.region = maxReg; });
                    }
                }

                if (checkAllRegionsConnected(simCells)) {
                    let areaCounts = [];
                    for(let i=0; i<=maxReg; i++) {
                        let count = simCells.filter(cell => cell.region === i).length;
                        if(count > 0) areaCounts.push(count * cellAreaM2);
                    }
                    let isHalfHalf = (areaCounts.length === 2 && areaCounts[0] === areaCounts[1]);
                    
                    if(areaCounts.length >= 2 && !isHalfHalf) {
                        resultBase = baseCells;
                        resultTargets = areaCounts.sort((a,b)=>a-b);
                        valid = true;
                    }
                }
            }
            return { cells: resultBase, targets: resultTargets, sideM: sideM, cellAreaM2: cellAreaM2 };
        }

        function generateBData() {
            let requiredTotal = 9;
            let requiredSub = 3;

            let subItemsInPool = globalItemsPool.filter(item => item.subWeight);
            if (globalItemsPool.length < requiredTotal || subItemsInPool.length < requiredSub) {
                globalItemsPool = [...masterItems].sort(() => 0.5 - Math.random());
            }

            let pickedSubs = [];
            let pickedNorms = [];
            
            for (let i = globalItemsPool.length - 1; i >= 0; i--) {
                if (globalItemsPool[i].subWeight && pickedSubs.length < requiredSub) {
                    pickedSubs.push(globalItemsPool.splice(i, 1)[0]);
                }
            }
            
            let neededRest = requiredTotal - pickedSubs.length;
            for (let i = 0; i < neededRest; i++) {
                if(globalItemsPool.length > 0) {
                    pickedNorms.push(globalItemsPool.shift());
                }
            }

            let pickedItems = [...pickedSubs, ...pickedNorms].sort(() => 0.5 - Math.random());
            let activeItems = pickedItems.map(item => ({ ...item, selected: false, subRemoved: false }));
            
            let numPicks = Math.floor(Math.random() * 3) + 3;
            let targetKg = 0; let picked = [];
            while(picked.length < numPicks) {
                let r = Math.floor(Math.random() * activeItems.length);
                if(!picked.includes(r)) picked.push(r);
            }
            
            let solArr = [];
            picked.forEach(idx => {
                let it = activeItems[idx];
                let w = it.weight;
                let subRemoved = false;
                if(it.subWeight && Math.random() > 0.5) { 
                    w -= it.subWeight; 
                    subRemoved = true; 
                }
                targetKg += w;
                solArr.push(subRemoved ? `${it.name}(卸下${it.subName})` : it.name);
            });
            
            return { items: activeItems, target: targetKg, solutionText: solArr.join(" + ") };
        }

        function undoPaint() {
            if(cellHistory.length > 1) {

        function setupCanvasInteraction() {
            const canvas = document.getElementById('landCanvas');
            if(!canvas) return;
            
            const paintCell = (e) => {
                const rect = canvas.getBoundingClientRect();
                let clientX = e.clientX || (e.touches && e.touches[0].clientX);
                let clientY = e.clientY || (e.touches && e.touches[0].clientY);
                let gridX = Math.floor((clientX - rect.left) / scaleSize);
                let gridY = Math.floor((clientY - rect.top) / scaleSize);
                
                let targetCell = currentCells.find(c => c.x === gridX && c.y === gridY);
                if(targetCell && targetCell.region !== activeBrush) {
                    targetCell.region = activeBrush;
                    drawPaintLand(); updateAreaDisplay();
                }
            };

            canvas.onmousedown = (e) => { e.preventDefault(); isPainting = true; paintCell(e); };
            canvas.onmousemove = (e) => { if(isPainting) { e.preventDefault(); paintCell(e); } };
            canvas.onmouseup = canvas.onmouseout = () => { 
                if(isPainting) { 
                    isPainting = false; 
                    cellHistory.push(JSON.parse(JSON.stringify(currentCells))); 
                } 
            };
            
            canvas.ontouchstart = (e) => { e.preventDefault(); isPainting = true; paintCell(e); };
            canvas.ontouchmove = (e) => { if(isPainting) { e.preventDefault(); paintCell(e); } };
            canvas.ontouchend = () => { if(isPainting) { isPainting = false; cellHistory.push(JSON.parse(JSON.stringify(currentCells))); } };
        }

        function drawPaintLand() {
            const canvas = document.getElementById('landCanvas');
            if(!canvas) return;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            currentCells.forEach(c => {
                if (c.region === -1) {
                    ctx.fillStyle = "rgba(255, 255, 255, 0.05)";
                } else {
                    ctx.fillStyle = regionColors[c.region];
                }
                ctx.fillRect(c.x * scaleSize, c.y * scaleSize, scaleSize, scaleSize);
                ctx.strokeStyle = "rgba(255,255,255,0.1)";
                ctx.strokeRect(c.x * scaleSize, c.y * scaleSize, scaleSize, scaleSize);
            });

            ctx.lineWidth = 2;
            currentCells.forEach(c => {
                if(c.region === -1) return; 

                ctx.strokeStyle = regionBorders[c.region];
                let top = !currentCells.find(o => o.x === c.x && o.y === c.y - 1 && o.region === c.region);
                let bottom = !currentCells.find(o => o.x === c.x && o.y === c.y + 1 && o.region === c.region);
                let left = !currentCells.find(o => o.x === c.x - 1 && o.y === c.y && o.region === c.region);
                let right = !currentCells.find(o => o.x === c.x + 1 && o.y === c.y && o.region === c.region);

                if(top) { ctx.beginPath(); ctx.moveTo(c.x*scaleSize, c.y*scaleSize); ctx.lineTo((c.x+1)*scaleSize, c.y*scaleSize); ctx.stroke(); }
                if(bottom) { ctx.beginPath(); ctx.moveTo(c.x*scaleSize, (c.y+1)*scaleSize); ctx.lineTo((c.x+1)*scaleSize, (c.y+1)*scaleSize); ctx.stroke(); }
                if(left) { ctx.beginPath(); ctx.moveTo(c.x*scaleSize, c.y*scaleSize); ctx.lineTo(c.x*scaleSize, (c.y+1)*scaleSize); ctx.stroke(); }
                if(right) { ctx.beginPath(); ctx.moveTo((c.x+1)*scaleSize, c.y*scaleSize); ctx.lineTo((c.x+1)*scaleSize, (c.y+1)*scaleSize); ctx.stroke(); }
            });
        }

        function updateAreaDisplay() {
            let areaMap = {};
            currentCells.forEach(c => { 
                if(c.region !== -1) areaMap[c.region] = (areaMap[c.region]||0) + 1; 
            });
            let areaVals = [];
            for(let r in areaMap) { areaVals.push(areaMap[r] * currentQData.data.cellAreaM2); }
            areaVals.sort((a,b)=>a-b);
            let areas = areaVals.map(v => formatBaseArea(v));
            document.getElementById('current-areas').innerText = areas.length > 0 ? areas.join("、") : "尚未塗色";
        }

        function checkAllRegionsConnected(cellsArray) {
            let regionMap = {};
            cellsArray.forEach(c => {
                if (c.region === -1) return;
                if(!regionMap[c.region]) regionMap[c.region] = [];
                regionMap[c.region].push(c);
            });
            for(let r in regionMap) {
                if(regionMap[r].length === 0) continue;
                let visited = new Set();
                let q = [regionMap[r][0]];
                visited.add(`${q[0].x},${q[0].y}`);
                while(q.length > 0) {
                    let curr = q.shift();
                    let neighbors = regionMap[r].filter(n => Math.abs(n.x - curr.x) + Math.abs(n.y - curr.y) === 1);
                    neighbors.forEach(n => {
                        let key = `${n.x},${n.y}`;
                        if(!visited.has(key)) { visited.add(key); q.push(n); }
                    });
                }
                if(visited.size !== regionMap[r].length) return false;
            }
            return true;
        }

        }
        function updateTotalWeight() {
            let total = 0;
            currentQData.data.items.forEach(it => {
                if(it.selected) total += (it.subRemoved ? it.weight - it.subWeight : it.weight);
            });
            let displaySum = total >= 1000 && total % 1000 === 0 ? `${total/1000} t (${total} kg)` : `${total} kg`;
            document.getElementById('current-sum').innerText = displaySum;
        }

        /* --- 驗證與過關 --- */
        function checkAnswer() {
            if(isPaused) return;
            
            if (currentQData.type === 'A') {
                if(currentCells.some(c => c.region === -1)) {
                    alert("尚有未塗色的區域喔！請將所有格子塗上對應的區塊顏色。");
                    return;
                }
                if(!checkAllRegionsConnected(currentCells)) {
                    alert("警告：相同顏色的區塊必須相連喔！請重新劃分塗色。");
                    return;
                }
            }

            let isCorrect = false;
            let correctDesc = "";

            if (currentQData.type === 'A') {
                let areaMap = {};
                currentCells.forEach(c => { areaMap[c.region] = (areaMap[c.region]||0) + 1; });
                
                let currentAreas = [];
                for(let r in areaMap) { currentAreas.push(areaMap[r] * currentQData.data.cellAreaM2); }
                let targetSorted = [...currentQData.data.targets].sort((a,b)=>a-b);
                currentAreas.sort((a,b)=>a-b);
                
                if (JSON.stringify(currentAreas) === JSON.stringify(targetSorted)) isCorrect = true;
                
                let targetText = currentQData.data.targets.map(a => formatBaseArea(a)).join("、");
                correctDesc = `目標面積應劃分為：<br><span style="color:var(--neon-green)">${targetText}</span> (可有多種切法)`;
            } else {
                let total = 0;
                currentQData.data.items.forEach(it => { if(it.selected) total += (it.subRemoved ? it.weight - it.subWeight : it.weight); });
                if (total === currentQData.data.target) isCorrect = true;
                correctDesc = `正確組合參考：<br><span style="color:var(--neon-green)">${currentQData.data.solutionText}</span>`;
            }

            const gameScreen = document.getElementById('game-screen');
            totalQuestionsPlayed++;

            if (isCorrect) {
                playBeep('correct');
                if (currentQData.type === 'A') correctA++; else correctB++;
                currentScore = correctA + correctB;
                
                document.getElementById('score-total').innerText = currentScore;
                
                gameScreen.style.boxShadow = "0 0 30px #00f0ff";
                setTimeout(() => gameScreen.style.boxShadow = "0 8px 32px 0 rgba(0,0,0,0.37)", 300);

                showFeedback(true, "太棒了！完全正確！✨", correctDesc);
            } else {
                playBeep('wrong');
                gameScreen.style.boxShadow = "0 0 30px #ff0055";
                setTimeout(() => gameScreen.style.boxShadow = "0 8px 32px 0 rgba(0,0,0,0.37)", 300);
                
                showFeedback(false, "哎呀！修復失敗 💥", correctDesc);
            }
        }
